SELECT z.Borough, z.Zone, count(*) AS trips, round(sum(t.total_amount), 2) AS revenue
               FROM (SELECT PULocationID, total_amount, trip_distance FROM trips_parquet WHERE pickup_date BETWEEN DATE'2024-04-01' AND DATE'2024-06-30' AND trip_distance > 0) t JOIN zones z ON t.PULocationID = z.LocationID
               GROUP BY 1, 2 ORDER BY revenue DESC LIMIT 10
