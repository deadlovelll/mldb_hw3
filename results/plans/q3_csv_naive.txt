               FROM (SELECT * FROM trips_csv WHERE tpep_pickup_datetime >= TIMESTAMP'2024-04-01 00:00:00' AND tpep_pickup_datetime < TIMESTAMP'2024-07-01 00:00:00' AND trip_distance > 0) t JOIN zones z ON t.PULocationID = z.LocationID
               GROUP BY 1, 2 ORDER BY revenue DESC LIMIT 10
